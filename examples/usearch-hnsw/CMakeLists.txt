cmake_minimum_required(VERSION 3.20)
project(zova_usearch LANGUAGES CXX)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
endif()
FetchContent_Declare(usearch
    URL https://codeload.github.com/unum-cloud/USearch/tar.gz/1e8a19253138510f35ee2c68dc641e5b5c83c7b5
    URL_HASH SHA256=298ced4f07dddd9e9d9f9228283da179c10664817c64b84497ec8a91ae1246aa)
FetchContent_Populate(usearch)
# The pin's noexcept key-map reservation defaults to throwing std::allocator.
# Use its own nullable allocator, matching the reserve() error contract. Keep
# this one-line compatibility patch in the build copy, never edit the archive.
file(COPY ${usearch_SOURCE_DIR}/include/usearch DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/include)
set(dense_header ${CMAKE_CURRENT_BINARY_DIR}/include/usearch/index_dense.hpp)
file(READ ${dense_header} dense_source)
set(old_lookup "flat_hash_multi_set_gt<key_and_slot_t, lookup_key_hash_t, lookup_key_same_t> slot_lookup_;")
string(FIND "${dense_source}" "${old_lookup}" lookup_position)
if(lookup_position EQUAL -1)
    message(FATAL_ERROR "USearch allocator compatibility patch does not match pinned source")
endif()
string(REPLACE "${old_lookup}" "flat_hash_multi_set_gt<key_and_slot_t, lookup_key_hash_t, lookup_key_same_t, dynamic_allocator_t> slot_lookup_;" dense_source "${dense_source}")
file(WRITE ${dense_header} "${dense_source}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_library(zova_usearch SHARED usearch.cpp)
target_include_directories(zova_usearch PRIVATE ../../include ${CMAKE_CURRENT_BINARY_DIR}/include)
target_compile_definitions(zova_usearch PRIVATE USEARCH_USE_NUMKONG=0 USEARCH_USE_OPENMP=0)
install(TARGETS zova_usearch LIBRARY DESTINATION usearch.zovaext RUNTIME DESTINATION usearch.zovaext)
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/extension.json CONTENT
    "{\"name\":\"usearch\",\"version\":\"1.0.0\",\"storage_prefix\":\"_zova_ext_usearch_\",\"zova_abi_min\":\"1.2.0\",\"capabilities\":\"\",\"library\":\"$<TARGET_FILE_NAME:zova_usearch>\",\"entrypoint\":\"zova_plugin_entry_v1\"}\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/extension.json DESTINATION usearch.zovaext)
install(FILES ${usearch_SOURCE_DIR}/LICENSE DESTINATION usearch.zovaext RENAME LICENSE-USearch)
install(FILES ../../LICENSE DESTINATION usearch.zovaext RENAME LICENSE-Zova)
install(FILES README.md DESTINATION usearch.zovaext)
set(ZOVA_C_ABI_LIBRARY "" CACHE FILEPATH "Zova static C ABI for external tests")
set(ZOVA_COMPILER_RT_LIBRARY "" CACHE FILEPATH "Generated-C host compiler runtime")
if(ZOVA_C_ABI_LIBRARY)
    find_package(Threads REQUIRED)
    add_executable(usearch_test ../../tests/usearch_hnsw.cpp)
    add_executable(usearch_faults ../../tests/usearch_hnsw_faults.cpp)
    target_include_directories(usearch_faults PRIVATE ../../include ${CMAKE_CURRENT_BINARY_DIR}/include)
    target_compile_definitions(usearch_faults PRIVATE USEARCH_USE_NUMKONG=0 USEARCH_USE_OPENMP=0)
    target_include_directories(usearch_test PRIVATE ../../include)
    target_link_libraries(usearch_test PRIVATE ${ZOVA_C_ABI_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
    if(ZOVA_COMPILER_RT_LIBRARY)
        target_link_libraries(usearch_test PRIVATE ${ZOVA_COMPILER_RT_LIBRARY})
    endif()
    if(WIN32)
        target_link_libraries(usearch_test PRIVATE ntdll)
    else()
        target_link_libraries(usearch_test PRIVATE m)
    endif()
endif()
